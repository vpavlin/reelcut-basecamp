cmake_minimum_required(VERSION 3.16)
project(reelcut_remote_test CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(nlohmann_json REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium REQUIRED)
add_executable(remote_test remote_test.cpp ../src/remote.cpp)
target_include_directories(remote_test PRIVATE ../src ${SODIUM_INCLUDE_DIR})
target_link_libraries(remote_test Qt6::Core Qt6::Network nlohmann_json::nlohmann_json ${SODIUM_LIBRARY})
